def sanitize_master_key(val):
    if isinstance(val, list):
        val = val[0] if val else "]"
    if not val or str(val).upper() in ["NONE", "NULL", ""]:
        return "]"
    k = str(val).upper()
    if k in KEY_TO_VK or k in SPECIAL_KEYS_MAP.values():
        return k
    return "]"

def save_data(filepath=CONFIG_FILE):
    save_dict = {
        "theme": ctk.get_appearance_mode(),
        "master_toggle_key": sanitize_master_key(master_toggle_key),
        "macros": {}
    }
    for mid, data in list(mappings.items()):
        save_dict["macros"][mid] = {
            "trigger": data["trigger"], "mode": data["mode"], "actions": data["actions"]
        }
    try:
        with open(filepath, "w", encoding="utf-8") as f: json.dump(save_dict, f, indent=4)
    except Exception: pass

# --- System Watchdog & Safety Protection ---
MAX_ACTIONS_PER_SEC = 50
MAX_LAG_TOLERANCE_SEC = 0.40

action_history = deque()
watchdog_lock = threading.Lock()

_sim_lock = threading.Lock()
_sim_depth = 0

def enter_sim():
    global _sim_depth
    with _sim_lock:
        _sim_depth += 1

def exit_sim():
    global _sim_depth
    with _sim_lock:
        _sim_depth = max(0, _sim_depth - 1)

def is_simulating():
    with _sim_lock:
        return _sim_depth > 0

CURRENT_PID = os.getpid()

_is_own_window_focused_cache = False

def _background_focus_tracker():
    global _is_own_window_focused_cache
    pid = wintypes.DWORD()
    while True:
        try:
            hwnd = user32.GetForegroundWindow()
            user32.GetWindowThreadProcessId(hwnd, ctypes.byref(pid))
            _is_own_window_focused_cache = (pid.value == CURRENT_PID)
        except Exception:
            _is_own_window_focused_cache = False
        time.sleep(0.15)

threading.Thread(target=_background_focus_tracker, daemon=True).start()

def parse_and_limit_delay(val_str):
    v = val_str.strip().lower()
    if v in ["0", "inf", "∞", "infinite"]: return 0
    try:
        parsed = int(v)
        if parsed < 0: return 0
        if parsed > 10000: return 10000
        return parsed
    except Exception:
        return 100

def format_delay_text(delay_ms):
    if delay_ms <= 0: return "∞ Inf"
    elif delay_ms >= 10000: return "10 s"
    elif delay_ms >= 1000: return f"{delay_ms / 1000:.1f} s"
    else: return f"{delay_ms} ms"

def validate_watchdog_safety():
    now = time.time()
    with watchdog_lock:
        while action_history and now - action_history[0] > 1.0:
            action_history.popleft()
        
        action_history.append(now)
        if len(action_history) > MAX_ACTIONS_PER_SEC:
            release_all_active_macros()
            set_status_indicator("● OVERLOAD STOPPED", DANGER_NEON)
            return False
    return True

# --- Direct Win32 Low-Level Hardware Key Executor ---
def execute_key_action(key_str, action_type):
    if not key_str or not validate_watchdog_safety(): return

    key_upper = key_str.strip().upper()
    vk = KEY_TO_VK.get(key_upper)
    if not vk: return

    scan_code = user32.MapVirtualKeyW(vk, 0)
    extended = 1 if vk in [0x21, 0x22, 0x23, 0x24, 0x25, 0x26, 0x27, 0x28, 0x2D, 0x2E] else 0

    enter_sim()
    try:
        if action_type == "Press":
            user32.keybd_event(vk, scan_code, extended, 0)
            time.sleep(0.015)
            user32.keybd_event(vk, scan_code, extended | 2, 0)
        elif action_type == "Hold":
            user32.keybd_event(vk, scan_code, extended, 0)
        elif action_type in ["Unhold", "Release"]:
            user32.keybd_event(vk, scan_code, extended | 2, 0)
    except Exception:
        pass
    finally:
        time.sleep(0.002)
        exit_sim()

def release_all_keys_for_macro(mid):
    m_data = mappings.get(mid)
    if not m_data: return
    enter_sim()
    try:
        for act in m_data["actions"]:
            if not act.get("key"): continue
            key_upper = act["key"].strip().upper()
            vk = KEY_TO_VK.get(key_upper)
            if vk:
                scan_code = user32.MapVirtualKeyW(vk, 0)
                extended = 1 if vk in [0x21, 0x22, 0x23, 0x24, 0x25, 0x26, 0x27, 0x28, 0x2D, 0x2E] else 0
                try:
                    user32.keybd_event(vk, scan_code, extended | 2, 0)
                except Exception:
                    pass
    finally:
        time.sleep(0.002)
        exit_sim()

def release_all_active_macros():
    for mid, data in list(mappings.items()):
        data["is_running"] = False
        release_all_keys_for_macro(mid)
        execution_locks[mid] = False

def run_sequence(mid):
    m_data = mappings.get(mid)
    if not m_data: return
    executed_once_keys = set()
    
    try:
        set_status_indicator("● MACRO ACTIVE", CYAN_NEON)
        while m_data["is_running"] and script_enabled:
            for idx, act in enumerate(m_data["actions"]):
                if not m_data["is_running"] or not script_enabled: break
                
                loop_type = act.get("loop", "Repeat")
                if loop_type == "One-Time" and idx in executed_once_keys:
                    continue
                    
                act_type = act.get("type", "Press")
                
                start_step_time = time.time()
                execute_key_action(act["key"], act_type)
                executed_once_keys.add(idx)
                
                if not m_data["is_running"] or not script_enabled: break

                delay_ms = act["delay"]
                if delay_ms <= 0:
                    time.sleep(0.001)
                else:
                    delay_sec = delay_ms / 1000.0
                    elapsed = 0
                    while elapsed < delay_sec and m_data["is_running"] and script_enabled:
                        time.sleep(0.005)
                        elapsed += 0.005
                        
                actual_spent = time.time() - start_step_time
                expected_spent = (delay_ms / 1000.0) + 0.015
                if actual_spent - expected_spent > MAX_LAG_TOLERANCE_SEC:
                    release_all_active_macros()
                    set_status_indicator("● LAG DETECTED - STOPPED", DANGER_NEON)
                    return

            if m_data["mode"] == "Press (1x)":
                m_data["is_running"] = False
                break
    finally:
        release_all_keys_for_macro(mid)
        execution_locks[mid] = False
        if script_enabled and not any(m.get("is_running", False) for m in mappings.values()):
            set_status_indicator("● SYSTEM READY", GREEN_NEON)

def trigger_mappings(key_name, is_press):
    if not script_enabled: return
    for mid, data in list(mappings.items()):
        if data["trigger"] == key_name:
            if is_press:
                if execution_locks.get(mid, False): continue
                
                if data["mode"] == "Hold" and not data["is_running"]:
                    data["is_running"] = True
                    execution_locks[mid] = True
                    threading.Thread(target=run_sequence, args=(mid,), daemon=True).start()
                elif data["mode"] in ["Toggle", "Press (1x)"]:
                    data["is_running"] = not data["is_running"]
                    if data["is_running"]:
                        execution_locks[mid] = True
                        threading.Thread(target=run_sequence, args=(mid,), daemon=True).start()
                    else:
                        release_all_keys_for_macro(mid)
            else:
                if data["mode"] == "Hold":
                    data["is_running"] = False
                    release_all_keys_for_macro(mid)

def set_recording(rectype, mid):
    recording_state["type"] = rectype
    recording_state["id"] = mid
    refresh_ui()

def process_record(key_name):
    if recording_state["type"] == "trigger":
        mappings[recording_state["id"]]["trigger"] = None if key_name == "ESC" else key_name
    elif recording_state["type"] == "action":
        if key_name == "ESC":
            temp_action_keys.pop(recording_state["id"], None)
        else:
            temp_action_keys[recording_state["id"]] = key_name

    recording_state["type"] = None
    recording_state["id"] = None
    save_data()
    app.after(0, refresh_ui)

# --- Physical Hardware Key Mapper ---
def get_key_name(key):
    if key in SPECIAL_KEYS_MAP:
        return SPECIAL_KEYS_MAP[key]

    vk = getattr(key, 'vk', None)
    if vk is not None and vk in VK_TO_KEY:
        return VK_TO_KEY[vk]

    if hasattr(key, 'char') and key.char:
        c = key.char.strip()
        if c in ARABIC_TO_ENGLISH_MAP:
            return ARABIC_TO_ENGLISH_MAP[c]
        c_upper = c.upper()
        if c_upper in KEY_TO_VK:
            return c_upper

    if hasattr(key, 'name') and key.name:
        n = key.name.upper()
        if n in KEY_TO_VK:
            return n

    return None

def is_master_toggle_pressed(key, key_name):
    target_str = str(master_toggle_key).upper()
    if key_name and key_name.upper() == target_str:
        return True
    
    target_vk = KEY_TO_VK.get(target_str)
    current_vk = getattr(key, 'vk', None)
    if target_vk and current_vk and target_vk == current_vk:
        return True
    return False

# --- Pure Dual-Language Keyboard Listeners ---
def on_press(key):
    global script_enabled, master_toggle_key
    if is_simulating(): return
    key_name = get_key_name(key)

    # 1. Master Toggle Recording Priority
    if recording_state["type"] == "master_toggle":
        if key_name and key_name != "ESC":
            master_toggle_key = sanitize_master_key(key_name)
        recording_state["type"] = None
        recording_state["id"] = None
        save_data()
        app.after(0, refresh_ui)
        return

    # 2. Global Master Pause Toggle Check
    if is_master_toggle_pressed(key, key_name) and not recording_state["type"]:
        script_enabled = not script_enabled
        if not script_enabled:
            release_all_active_macros()
            set_status_indicator("● SCRIPT PAUSED (BROWSING MODE)", DANGER_NEON)
        else:
            set_status_indicator("● SYSTEM READY", GREEN_NEON)
        app.after(0, update_guide_master_text)
        return

    if not script_enabled or not key_name: return

    if recording_state["type"]:
        process_record(key_name)
        return

    if _is_own_window_focused_cache: return
    threading.Thread(target=trigger_mappings, args=(key_name, True), daemon=True).start()

def on_release(key):
    if not script_enabled: return
    key_name = get_key_name(key)
    if not key_name or recording_state["type"]: return
    
    threading.Thread(target=trigger_mappings, args=(key_name, False), daemon=True).start()

keyboard.Listener(on_press=on_press, on_release=on_release).start()

def edit_action_node(mid, idx):
    act = mappings[mid]["actions"][idx]
    
    edit_win = ctk.CTkToplevel(app)
    edit_win.title("Edit Node")
    edit_win.geometry("300x260")
    edit_win.resizable(False, False)
    edit_win.configure(fg_color=COLOR_BG)
    edit_win.grab_set()

    ctk.CTkLabel(edit_win, text=f"Edit Action: {act['key']}", font=FONT_TITLE, text_color=COLOR_TEXT_MAIN).pack(pady=(15, 10))

    type_cb = ctk.CTkComboBox(edit_win, values=["Press", "Hold", "Unhold", "Release"], width=150,
                              fg_color=COLOR_CARD, text_color=COLOR_TEXT_MAIN,
                              dropdown_fg_color=COLOR_CARD, dropdown_text_color=COLOR_TEXT_MAIN)
    type_cb.set(act.get("type", "Press"))
    type_cb.pack(pady=5)

    loop_cb = ctk.CTkComboBox(edit_win, values=["Repeat", "One-Time"], width=150,
                              fg_color=COLOR_CARD, text_color=COLOR_TEXT_MAIN,
                              dropdown_fg_color=COLOR_CARD, dropdown_text_color=COLOR_TEXT_MAIN)
    loop_cb.set(act.get("loop", "Repeat"))
    loop_cb.pack(pady=5)

    delay_e = ctk.CTkEntry(edit_win, width=150, placeholder_text="Delay (ms)", fg_color=COLOR_CARD, text_color=COLOR_TEXT_MAIN)
    delay_e.insert(0, str(act["delay"]))
    delay_e.pack(pady=5)

    def save_node_changes():
        act["type"] = type_cb.get()
        act["loop"] = loop_cb.get()
        act["delay"] = parse_and_limit_delay(delay_e.get())
        save_data()
        refresh_ui()
        edit_win.destroy()

    ctk.CTkButton(edit_win, text="Save Changes", fg_color=CYAN_NEON, text_color="#000", font=FONT_BTN, command=save_node_changes).pack(pady=15)

def remove_action_node(mid, idx):
    mappings[mid]["actions"].pop(idx)
    save_data()
    refresh_ui()

def delete_mapping(mid):
    del mappings[mid]
    save_data()
    refresh_ui()

def update_mode(mid, val):
    mappings[mid]["mode"] = val
    save_data()

def add_action_node(mid, act_type_cb, loop_type_cb, delay_entry):
    key = temp_action_keys.get(mid)
    if not key:
        messagebox.showwarning("Warning", "Please select a key first!")
        return
    act_type = act_type_cb.get()
    loop_type = loop_type_cb.get()
    delay = parse_and_limit_delay(delay_entry.get())
    
    mappings[mid]["actions"].append({
        "key": key, "type": act_type, "loop": loop_type, "delay": delay
    })
    temp_action_keys.pop(mid, None)
    save_data()
    refresh_ui()

def refresh_ui():
    update_master_btn_ui()
    update_guide_master_text()
    for w in scroll_area.winfo_children(): w.destroy()
    
    for mid, data in list(mappings.items()):
        card = ctk.CTkFrame(scroll_area, fg_color=COLOR_CARD, border_color=COLOR_BORDER, border_width=1, corner_radius=12, height=150)
